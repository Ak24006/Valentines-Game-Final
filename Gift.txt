<body>
    <div id="game-container">
        <h2 id="question">Step 1: Pick a surprise gift for me!</h2>
        <div class="options">
            <button onclick="nextStep()">A bouquet of roses 🌹</button>
            <button onclick="nextStep()">A box of chocolates 🍫</button>
            <button onclick="nextStep()">A cute teddy bear 🧸</button>
        </div>
    </div>
    
    <script>
        let step = 1;
        const questions = [
            "Step 2: Choose the perfect place for our date!",
            "Step 3: Pick a song for our first dance!",
            "Step 4: How would you express your love?",
            "Final Step: The big question..."
        ];
        const options = [
            ["A romantic rooftop dinner 🍽️", "A cozy movie night 🎬", "A sunset beach walk 🏖️"],
            ["Perfect by Ed Sheeran 🎶", "Thinking Out Loud 🎵", "Can’t Help Falling in Love 💕"],
            ["With a heartfelt letter 💌", "A surprise date plan 💕", "Just looking into your eyes 😍"]
        ];

        function nextStep() {
            if (step < questions.length) {
                document.getElementById('question').innerText = questions[step];
                let btnContainer = document.querySelector('.options');
                btnContainer.innerHTML = '';
                options[step - 1].forEach(option => {
                    let btn = document.createElement('button');
                    btn.innerText = option;
                    btn.onclick = () => nextStep();
                    btnContainer.appendChild(btn);
                });
                step++;
            } else {
                document.getElementById('game-container').innerHTML = `
                    <h2>Will you be my Valentine? ❤️</h2>
                    <button id='yesButton' onclick='finalResponse()'>Yes, of course! 💖</button>
                `;
            }
        }

        function finalResponse() {
            document.getElementById('game-container').innerHTML = `
                <h2>Yay! You’re my Valentine! 💖🎉</h2>
                <p>Sorry Piluu, maaf kardo, ho jata hai late mere se! 😘

                    I Love You💖</p>
            `;
        }
    </script>
